# Strict .specify version-control policy.
#
# Agent command directories live OUTSIDE .specify/ and are NOT covered here:
# add assets/agent-commands-gitignore.txt to the ROOT .gitignore instead.
#
# Two safe placements (see references/vcs-policy.md):
# - Append to the managed .specify/.gitignore (patterns relative to .specify/).
#   Only durable if `specify init` is never re-run: `init --force` rewrites that
#   file and drops appended rules (observed on 1.0.13).
# - Put the rules in the root .gitignore with every pattern prefixed ".specify/"
#   (e.g. ".specify/presets/*"). Required when init is part of the restore path.

# --- Installed from catalogs. Reconstructed with `specify bundle install`. ---
extensions/*/*
!extensions/*/*-config.yml
presets/*
workflows/*
!workflows/overlays/
workflows/steps/*

# Keep machine-local extension overrides ignored even though the negation above
# would otherwise match them (*-config.yml also matches local-config.yml).
extensions/*/local-config.yml
extensions/*/*-config.local.yml

# --- Runtime state, caches, and backups. ---
workflows/runs/
workflows/.cache/
extensions/.cache/
extensions/.backup/

# Workflow installs take an flock at the .specify/ root and never unlink it.
.workflow-install.lock

# --- Project-owned bundle source (.specify/bundles/<id>/) is intent: bundle.yml
# --- and its README.md stay tracked. Only the built artifact is derived output. ---
bundles/*/*.zip

# --- Installation-state records. Rebuilt on reconstruction; stale copies can
# --- make the bundler believe an ignored component is present. ---
extensions/.registry
bundle-records.json
workflows/workflow-registry.json
